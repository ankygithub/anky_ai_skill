12.7.2 DICOM解析示例

import pydicom
import numpy as np
from pydicom.uid import UID

def parse_dicom_series(dicom_dir):
    """解析DICOM序列文件，返回三维影像数组和元数据"""
    slices = []
    for f in glob.glob(f"{dicom_dir}/*.dcm"):
        ds = pydicom.dcmread(f)
        
        # 提取关键元数据
        metadata = {
            "patient_id": ds.get("PatientID", ""),
            "study_date": ds.get("StudyDate", ""),
            "modality": ds.get("Modality", ""),
            "slice_thickness": float(ds.get("SliceThickness", 1.0)),
            "pixel_spacing": [float(x) for x in ds.get("PixelSpacing", [1, 1])],
            "image_position": [float(x) for x in ds.get("ImagePositionPatient", [0, 0, 0])],
            "window_center": float(ds.get("WindowCenter", 40)),
            "window_width": float(ds.get("WindowWidth", 400)),
        }
        
        # 提取像素数据并转换为HU值（CT）
        pixel_array = ds.pixel_array.astype(np.float32)
        if hasattr(ds, 'RescaleSlope') and hasattr(ds, 'RescaleIntercept'):
            pixel_array = pixel_array * ds.RescaleSlope + ds.RescaleIntercept
        
        slices.append({
            "data": pixel_array,
            "position": metadata["image_position"][2],
            "metadata": metadata
        })
    
    # 按Z轴位置排序
    slices.sort(key=lambda x: x["position"])
    volume = np.stack([s["data"] for s in slices], axis=-1)
    
    return volume, slices[0]["metadata"]

12.7.3 RAG检索逻辑示例

from langchain.vectorstores import Milvus
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.llms import VLLM

class MedicalRAG:
    def __init__(self, milvus_uri, model_path):
        self.embeddings = HuggingFaceEmbeddings(
            model_name="shibing624/text2vec-base-chinese"
        )
        self.vectorstore = Milvus(
            embedding_function=self.embeddings,
            connection_args={"uri": milvus_uri},
            collection_name="medical_knowledge"
        )
        self.llm = VLLM(model=model_path, temperature=0.1)
    
    def query(self, question: str, department: str = None) -> dict:
        """医疗知识问答"""
        # 1. 混合检索
        search_filter = {"department": department} if department else {}
        docs = self.vectorstore.similarity_search(
            question, k=5, filter=search_filter
        )
        
        # 2. 构建Prompt（带来源引用）
        context = "\n\n".join([
            f"[来源{i+1}] {doc.metadata['source']}\n{doc.page_content}"
            for i, doc in enumerate(docs)
        ])
        
        prompt = f"""基于以下医学知识回答问题。如果知识不足以回答，请明确告知。
        
参考知识：
{context}

问题：{question}

回答（请附带参考来源编号）："""
        
        # 3. 生成答案
        answer = self.llm(prompt)
        
        # 4. 返回结果（含来源引用）
        return {
            "question": question,
            "answer": answer,
            "sources": [
                {
                    "title": doc.metadata.get("title", ""),
                    "source": doc.metadata.get("source", ""),
                    "evidence_level": doc.metadata.get("evidence_level", "")
                }
                for doc in docs
            ]
        }
