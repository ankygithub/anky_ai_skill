4.9.3 意图识别接口伪代码

以下是一个基于 BERT 微调的意图识别接口示例（Python）：

from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch
import json

class IntentClassifier:
    """基于BERT微调的意图识别器"""
    
    def __init__(self, model_path: str, device: str = "cuda"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_path)
        self.device = torch.device(device if torch.cuda.is_available() else "cpu")
        self.model.to(self.device)
        self.model.eval()
        
        # 意图映射
        self.intent_labels = [
            "order_query", "shipping_query", "refund_apply", 
            "return_apply", "product_consult", "complaint",
            "account_issue", "payment_issue", "invoice_issue"
        ]
        
    def predict(self, text: str, top_k: int = 3) -> dict:
        """
        预测用户意图
        
        Args:
            text: 用户输入文本
            top_k: 返回Top-K个候选意图
            
        Returns:
            {
                "primary_intent": "order_query",
                "confidence": 0.89,
                "top_k": [
                    {"intent": "order_query", "confidence": 0.89},
                    {"intent": "shipping_query", "confidence": 0.07},
                    {"intent": "refund_apply", "confidence": 0.03}
                ]
            }
        """
        inputs = self.tokenizer(
            text, 
            truncation=True, 
            max_length=128, 
            padding="max_length",
            return_tensors="pt"
        ).to(self.device)
        
        with torch.no_grad():
            outputs = self.model(**inputs)
            logits = outputs.logits
            probabilities = torch.softmax(logits, dim=-1)
            
        # 获取Top-K
        top_probs, top_indices = torch.topk(probabilities[0], top_k)
        
        results = []
        for prob, idx in zip(top_probs, top_indices):
            results.append({
                "intent": self.intent_labels[idx.item()],
                "confidence": round(prob.item(), 4)
            })
            
        return {
            "primary_intent": results[0]["intent"],
            "confidence": results[0]["confidence"],
            "top_k": results
        }

# 使用示例
classifier = IntentClassifier("./models/intent-bert-v2")
result = classifier.predict("我的订单什么时候发货？")
print(json.dumps(result, ensure_ascii=False, indent=2))

以下是一个基于 LLM Few-shot 的意图识别接口示例：

import openai
import json

class LLMIntentClassifier:
    """基于大模型的意图识别器（Few-shot方式）"""
    
    SYSTEM_PROMPT = """你是一个电商智能客服的意图识别助手。
请根据用户输入，识别其咨询意图。

可选意图类别：
- order_query: 订单查询
- shipping_query: 物流咨询
- refund_apply: 退款申请
- return_apply: 退换货申请
- product_consult: 商品咨询
- complaint: 投诉建议
- account_issue: 账户问题
- payment_issue: 支付问题
- invoice_issue: 发票问题

请仅返回JSON格式：
{"primary_intent": "意图类别", "confidence": 0.xx}"""
    
    def __init__(self, api_key: str, model: str = "qwen-max"):
        self.client = openai.OpenAI(
            api_key=api_key,
            base_url="https://dashscope.aliyuncs.com/compatible-mode/v1"
        )
        self.model = model
        
    def predict(self, text: str) -> dict:
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": self.SYSTEM_PROMPT},
                {"role": "user", "content": text}
            ],
            temperature=0.1,  # 低温度确保稳定输出
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        return result

# 使用示例
classifier = LLMIntentClassifier(api_key="your-api-key")
result = classifier.predict("我昨天买的手机还没发货，能帮我查查吗？")
print(result)  # {"primary_intent": "order_query", "confidence": 0.92}

Redis缓存读写封装

import redis
import json
from typing import Optional

class SessionCache:
    """会话状态Redis缓存管理"""
    
    def __init__(self, redis_host: str, redis_port: int = 6379, db: int = 0):
        self.redis = redis.Redis(
            host=redis_host, 
            port=redis_port, 
            db=db,
            decode_responses=True,
            socket_timeout=0.5,
            socket_connect_timeout=0.5
        )
        self.TTL = 1800  # 30分钟
        
    def save_session(self, session_id: str, session_data: dict) -> bool:
        """保存会话状态"""
        try:
            key = f"session:{session_id}"
            self.redis.setex(
                key,
                self.TTL,
                json.dumps(session_data, ensure_ascii=False)
            )
            return True
        except redis.RedisError as e:
            # 缓存写入失败不应影响主流程
            print(f"Redis写入失败: {e}")
            return False
            
    def get_session(self, session_id: str) -> Optional[dict]:
        """获取会话状态"""
        try:
            key = f"session:{session_id}"
            data = self.redis.get(key)
            if data:
                return json.loads(data)
            return None
        except redis.RedisError as e:
            print(f"Redis读取失败: {e}")
            return None
            
    def delete_session(self, session_id: str) -> bool:
        """删除会话状态"""
        try:
            key = f"session:{session_id}"
            return self.redis.delete(key) > 0
        except redis.RedisError as e:
            print(f"Redis删除失败: {e}")
            return False

    def update_session_field(self, session_id: str, field: str, value) -> bool:
        """更新会话中的某个字段（避免全量读写）"""
        try:
            key = f"session:{session_id}"
            # 使用HSET实现字段级别更新
            self.redis.hset(key, field, json.dumps(value, ensure_ascii=False))
            self.redis.expire(key, self.TTL)  # 刷新TTL
            return True
        except redis.RedisError as e:
            print(f"Redis更新失败: {e}")
            return False

熔断器实现（业务系统调用保护）

import time
from enum import Enum
from typing import Callable, Any

class CircuitState(Enum):
    CLOSED = "closed"       # 正常状态
    OPEN = "open"           # 熔断状态
    HALF_OPEN = "half_open" # 半开状态

class CircuitBreaker:
    """熔断器实现 - 保护业务系统调用"""
    
    def __init__(
        self,
        name: str,
        failure_threshold: int = 5,    # 连续失败5次触发熔断
        recovery_timeout: int = 30,     # 30秒后尝试恢复
        half_open_max_calls: int = 1    # 半开状态允许的最大调用数
    ):
        self.name = name
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0
        self.half_open_calls = 0
        
    def execute(self, func: Callable, *args, **kwargs) -> Any:
        """执行受保护的调用"""
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
            else:
                raise Exception(f"[{self.name}] 熔断中，请稍后重试")
                
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e
            
    def _on_success(self):
        """成功回调"""
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.CLOSED
            self.failure_count = 0
        elif self.state == CircuitState.CLOSED:
            self.failure_count = 0
            
    def _on_failure(self):
        """失败回调"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.OPEN
        elif self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            print(f"[{self.name}] 触发熔断，将在 {self.recovery_timeout}s 后尝试恢复")

架构师提醒

示例代码仅用于说明核心设计思路，生产环境使用时需要考虑：

  • 错误处理：示例中的异常处理较简单，生产环境需要更完善的错误分类和重试策略
  • 日志记录：所有关键操作需要记录结构化日志，便于追踪和分析
  • 监控告警：熔断器状态、缓存命中率、接口延迟等指标需要接入监控系统
  • 安全加固：API密钥、数据库密码等敏感信息必须通过密钥管理服务获取
  • 并发安全：熔断器在高并发场景下需要加锁或使用原子操作
