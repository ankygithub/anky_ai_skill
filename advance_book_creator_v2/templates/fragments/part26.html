示例一：RAG管道伪代码

class RAGPipeline:
    """完整的RAG管道实现"""
    
    def __init__(self, config: dict):
        self.embedder = EmbeddingModel(config["embedding_model"])
        self.vector_db = MilvusClient(config["milvus_url"])
        self.reranker = RerankerModel(config["reranker_model"])
        self.llm = LLMClient(config["llm_model"])
        self.max_recall = config.get("max_recall", 20)
        self.top_k = config.get("top_k", 5)
        self.similarity_threshold = config.get("similarity_threshold", 0.5)
        
    def ingest_document(self, doc: Document) -> None:
        """文档入库"""
        # Step 1: 文档解析
        parsed = self.parse_document(doc)
        
        # Step 2: 智能切片
        chunks = self.chunk_document(parsed)
        
        # Step 3: Embedding
        vectors = self.embedder.encode([c.text for c in chunks])
        
        # Step 4: 索引（附带ACL元数据）
        for chunk, vector in zip(chunks, vectors):
            self.vector_db.insert({
                "vector": vector,
                "text": chunk.text,
                "doc_id": doc.id,
                "chunk_id": chunk.id,
                "permissions": doc.permissions,
                "metadata": chunk.metadata
            })
            
    def query(self, question: str, user_id: str) -> dict:
        """用户查询"""
        # Step 1: 查询改写
        rewritten = self.rewrite_query(question)
        
        # Step 2: 向量化
        query_vector = self.embedder.encode(rewritten)
        
        # Step 3: 带权限过滤的检索
        filter_expr = self.build_permission_filter(user_id)
        candidates = self.vector_db.search(
            vector=query_vector,
            filter=filter_expr,
            top_k=self.max_recall
        )
        
        # Step 4: 重排
        reranked = self.reranker.rank(
            query=question,
            documents=[c["text"] for c in candidates]
        )
        
        # Step 5: 取Top-K
        top_docs = reranked[:self.top_k]
        
        # Step 6: 生成答案
        answer = self.llm.generate(
            prompt=self.build_prompt(question, top_docs),
            temperature=0.1
        )
        
        # Step 7: 附加引用
        result = {
            "answer": answer.text,
            "sources": [
                {
                    "doc_id": d["doc_id"],
                    "text": d["text"],
                    "score": d["score"]
                }
                for d in top_docs
            ]
        }
        return result

示例二：多路召回配置

class MultiRecallStrategy:
    """多路召回策略实现"""
    
    def __init__(self, config: dict):
        # 向量检索
        self.vector_search = VectorSearch(
            embedding_model=config["embedding_model"],
            vector_db=config["vector_db"],
            top_k=config.get("vector_top_k", 15)
        )
        
        # 关键词检索（BM25）
        self.keyword_search = KeywordSearch(
            es_url=config["elasticsearch_url"],
            analyzer=config.get("analyzer", "ik_max_word"),
            top_k=config.get("keyword_top_k", 10)
        )
        
        # 融合策略
        self.fusion_method = config.get("fusion", "rrf")  # rrf / linear_weighted
        
    def search(self, query: str, user_id: str) -> list[dict]:
        """执行多路召回"""
        # 并行执行两路召回
        import concurrent.futures
        with concurrent.futures.ThreadPoolExecutor(max_workers=2) as executor:
            future_vector = executor.submit(
                self.vector_search.search, query, user_id
            )
            future_keyword = executor.submit(
                self.keyword_search.search, query, user_id
            )
            
            vector_results = future_vector.result()
            keyword_results = future_keyword.result()
        
        # 融合排序
        if self.fusion_method == "rrf":
            return self.reciprocal_rank_fusion(
                [vector_results, keyword_results],
                top_k=20
            )
        elif self.fusion_method == "linear_weighted":
            return self.linear_weighted_fusion(
                vector_results, keyword_results,
                alpha=0.6,  # 偏向向量检索
                top_k=20
            )
            
    def reciprocal_rank_fusion(
        self, 
        result_lists: list[list[dict]], 
        top_k: int,
        k: int = 60  # RRF常数
    ) -> list[dict]:
        """倒数排名融合"""
        scores = {}
        for results in result_lists:
            for rank, item in enumerate(results):
                doc_id = item["doc_id"]
                rrf_score = 1.0 / (k + rank + 1)
                scores[doc_id] = scores.get(doc_id, 0) + rrf_score
                
        # 按RRF得分排序
        sorted_docs = sorted(
            scores.items(), key=lambda x: x[1], reverse=True
        )
        return sorted_docs[:top_k]

示例三：查询改写逻辑

class QueryRewriter:
    """查询改写器"""
    
    def __init__(self, llm: LLMClient):
        self.llm = llm
        
    def rewrite(self, question: str, strategy: str = "auto") -> str:
        """根据策略改写查询"""
        if strategy == "auto":
            # 自动判断最佳策略
            strategy = self.detect_strategy(question)
            
        if strategy == "clarification":
            return self.intent_clarification(question)
        elif strategy == "decomposition":
            return self.question_decomposition(question)
        elif strategy == "hyde":
            return self.hyde_generation(question)
        else:
            return question  # 不改写
            
    def intent_clarification(self, question: str) -> str:
        """意图澄清改写"""
        prompt = f"""
        请将以下用户问题改写为更适合知识库检索的形式。
        要求：
        1. 提取核心关键词
        2. 补充隐含的业务概念
        3. 保持语义不变
        
        用户问题：{question}
        改写结果：
        """
        return self.llm.generate(prompt, temperature=0.1).strip()
        
    def hyde_generation(self, question: str) -> str:
        """HyDE假设文档生成"""
        prompt = f"""
        请针对以下问题，生成一段可能出现在企业知识库中的参考答案。
        答案应该专业、准确，长度约100-200字。
        
        问题：{question}
        假设答案：
        """
        return self.llm.generate(prompt, temperature=0.3).strip()
        
    def detect_strategy(self, question: str) -> str:
        """自动检测适合的改写策略"""
        # 简单规则判断（实际可用LLM判断）
        if len(question) < 10:
            return "hyde"  # 简短问题用HyDE
        if "和" in question or "分别" in question or "各" in question:
            return "decomposition"  # 并列问题需要分解
        return "clarification"  # 默认意图澄清

示例四：权限过滤逻辑

class PermissionFilter:
    """权限过滤器"""
    
    def __init__(self, permission_service):
        self.permission_service = permission_service
        
    def build_filter_expr(self, user_id: str) -> str:
        """构建Milvus/ES权限过滤表达式"""
        # 获取用户权限信息
        user_info = self.permission_service.get_user_info(user_id)
        roles = user_info.get("roles", [])
        departments = user_info.get("departments", [])
        
        # 构建过滤表达式（Milvus语法）
        # 权限字段包含用户角色或部门，或为公开文档
        permission_conditions = []
        for role in roles:
            permission_conditions.append(f'permissions == "{role}"')
        for dept in departments:
            permission_conditions.append(f'permissions == "{dept}"')
        permission_conditions.append('permissions == "public"')
        
        # 不在拒绝列表中
        filter_expr = f'({" || ".join(permission_conditions)}) && deny_list not in ["{user_id}"]'
        
        return filter_expr
        
    def filter_results(self, results: list[dict], user_id: str) -> list[dict]:
        """检索后二次确认（防御性检查）"""
        user_info = self.permission_service.get_user_info(user_id)
        allowed_permissions = set(
            user_info.get("roles", []) + 
            user_info.get("departments", []) + 
            ["public"]
        )
        
        filtered = []
        for item in results:
            doc_permissions = set(item.get("permissions", []))
            deny_list = item.get("deny_list", [])
            
            # 检查是否有权限
            has_permission = bool(doc_permissions & allowed_permissions)
            is_denied = user_id in deny_list
            
            if has_permission and not is_denied:
                filtered.append(item)
                
        return filtered
架构师提醒

示例代码仅用于说明核心设计思路，生产环境使用时需要考虑：

  • 错误处理：示例中省略了异常处理，生产环境需要完善的错误分类和重试策略
  • 日志记录：所有关键操作需要记录结构化日志，便于追踪和分析
  • 监控告警：Embedding延迟、检索延迟、LLM调用延迟等指标需要接入监控系统
  • 安全加固：API密钥、数据库密码等敏感信息必须通过密钥管理服务获取
  • 性能优化：Embedding和检索操作需要批量处理，LLM调用需要流式输出
